        function getFiltered() {
            let list = state.data;
            if (state.showMastered === true) list = list.filter(item => item.mastered);
            else if (state.showMastered === false) list = list.filter(item => !item.mastered);
            if (state.searchQuery.trim()) {
                const q = state.searchQuery.trim().toLowerCase();
                list = list.filter(item =>
                    item.q.toLowerCase().includes(q) ||
                    item.a.toLowerCase().includes(q) ||
                    item.chapter.includes(q) ||
                    item.sub.includes(q)
                );
            }
            return list;
        }

        function updateFiltered() {
            state.filtered = getFiltered();
            renderOutline();
            updateStats();
            updateJumpOptions();
            updateFilterInfo();
            renderReviewList();
        }

        function updateStats() {
            const total = state.filtered.length;
            const done = state.filtered.filter(item => item.mastered).length;
            const pct = total === 0 ? 0 : Math.round((done / total) * 100);
            totalNum.textContent = total;
            doneNum.textContent = done;
            pctNum.textContent = pct + '%';
            progressFill.style.width = pct + '%';
            progressText.textContent = `${done} / ${total} 已掌握`;
            progressPct.textContent = pct + '%';
            footerTotal.textContent = total;
            const allQ = state.allQuestions.length;
            $('footerQuizCount').textContent = allQ;
        }

        function updateFilterInfo() {
            let label = '';
            if (state.showMastered === true) label = '✅ 已掌握';
            else if (state.showMastered === false) label = '📖 未掌握';
            else label = '全部';
            if (state.searchQuery.trim()) label += ' + 搜索';
            filterInfo.textContent = label;
        }

        function updateJumpOptions() {
            const chapters = [...new Set(state.data.map(d => d.chapter))];
            const current = chapterJump.value;
            chapterJump.innerHTML = '<option value="all">📖 全部章节</option>';
            chapters.forEach(ch => {
                const opt = document.createElement('option');
                opt.value = ch;
                opt.textContent = ch;
                chapterJump.appendChild(opt);
            });
            if (current && chapters.includes(current)) chapterJump.value = current;
        }

        // ---- 渲染大纲 ----
        function renderOutline() {
            const list = state.filtered;
            if (list.length === 0) {
                outlineEl.innerHTML = `<div class="empty-state"><div class="icon">📭</div><h3>没有匹配的知识点</h3><p>尝试调整筛选条件</p></div>`;
                return;
            }
            const chapterMap = new Map();
            list.forEach(item => {
                if (!chapterMap.has(item.chapter)) chapterMap.set(item.chapter, []);
                chapterMap.get(item.chapter).push(item);
            });
            const chapterOrder = [...new Set(state.data.map(d => d.chapter))];
            const sortedChapters = chapterOrder.filter(ch => chapterMap.has(ch));
            let html = '';
            let globalIdx = 0;
            sortedChapters.forEach((chapterName) => {
                const items = chapterMap.get(chapterName);
                const subMap = new Map();
                items.forEach(item => {
                    if (!subMap.has(item.sub)) subMap.set(item.sub, []);
                    subMap.get(item.sub).push(item);
                });
                const subOrder = [...new Set(items.map(i => i.sub))];
                const totalInChapter = items.length;
                const doneInChapter = items.filter(i => i.mastered).length;
                const isExpanded = state.expandedChapters.has(chapterName) || state.expandedChapters.size === 0;
                html += `
                    <div class="chapter" data-chapter="${chapterName}">
                        <div class="chapter-header" data-chapter="${chapterName}">
                            <div class="title">
                                ${chapterName}
                                <span class="badge">${totalInChapter} 条</span>
                                ${doneInChapter > 0 ? `<span class="badge done-badge">✅ ${doneInChapter}/${totalInChapter}</span>` : ''}
                            </div>
                            <span class="toggle-icon ${isExpanded ? 'open' : ''}">▾</span>
                        </div>
                        <div class="chapter-body ${isExpanded ? '' : 'collapsed'}">
                `;
                subOrder.forEach(subName => {
                    const subItems = subMap.get(subName);
                    html += `
                        <div class="subchapter">
                            <div class="subchapter-title">${subName} <span class="sub-badge">${subItems.length}</span></div>
                    `;
                    subItems.forEach((item) => {
                        const isMastered = item.mastered;
                        const qText = highlightText(item.q, state.searchQuery.trim());
                        const aText = highlightText(item.a, state.searchQuery.trim());
                        const hasStar = item.q.includes('★') || item.a.includes('★');
                        html += `
                            <div class="knowledge-item ${isMastered ? 'mastered' : ''}" data-id="${item.id}">
                                <span class="marker ${isMastered ? 'done' : ''}" data-id="${item.id}" title="点击标记掌握"></span>
                                <div class="content">
                                    <div class="q">${hasStar ? '<span class="star">★</span>' : ''}${qText}</div>
                                    <div class="a">${aText}</div>
                                </div>
                                <span class="item-index">#${++globalIdx}</span>
                            </div>
                        `;
                    });
                    html += `</div>`;
                });
                html += `</div></div>`;
            });
            outlineEl.innerHTML = html;
            // 事件绑定（与原版相同，省略）
            document.querySelectorAll('.chapter-header').forEach(header => {
                header.addEventListener('click', function(e) {
                    const ch = this.dataset.chapter;
                    const body = this.nextElementSibling;
                    const icon = this.querySelector('.toggle-icon');
                    if (body.classList.contains('collapsed')) {
                        body.classList.remove('collapsed');
                        icon.classList.add('open');
                        state.expandedChapters.add(ch);
                    } else {
                        body.classList.add('collapsed');
                        icon.classList.remove('open');
                        state.expandedChapters.delete(ch);
                    }
                });
            });
            document.querySelectorAll('.marker').forEach(marker => {
                marker.addEventListener('click', function(e) {
                    e.stopPropagation();
                    const id = parseInt(this.dataset.id);
                    const item = state.data.find(d => d.id === id);
                    if (item) {
                        item.mastered = !item.mastered;
                        saveStorage();
                        updateFiltered();
                    }
                });
            });
            document.querySelectorAll('.knowledge-item .content').forEach(el => {
                el.addEventListener('click', function(e) {
                    e.stopPropagation();
                    const marker = this.closest('.knowledge-item').querySelector('.marker');
                    if (marker) marker.click();
                });
            });
        }

        function highlightText(text, query) {
            if (!query || query.length < 1) return text;
            const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
            const regex = new RegExp(`(${escaped})`, 'gi');
            return text.replace(regex, '<span class="search-highlight">$1</span>');
        }

        // ---- 构建题库（仅种子+扩展坞） ----
        function buildFullQuiz() {
            let all = [];
            let zh = SEED_QUESTIONS.slice();
            let en = SEED_EN.slice();
            all = zh.concat(en);
            all.forEach((q, i) => q.id = i + 1);
            state.allQuestions = all;
            loadExtensionQuestions();
        }

        // ---- 扩展题库持久化 ----
        function loadExtensionQuestions() {
            try {
                const ext = localStorage.getItem('extensionQuestions');
                if (ext) {
                    const parsed = JSON.parse(ext);
                    if (Array.isArray(parsed) && parsed.length) {
                        const existingIds = new Set(state.allQuestions.map(q => q.id));
                        let maxId = Math.max(...state.allQuestions.map(q => q.id), 0);
                        parsed.forEach(q => {
                            if (!existingIds.has(q.id)) {
                                q.id = ++maxId;
                                state.allQuestions.push(q);
                                existingIds.add(q.id);
                            }
                        });
                        saveExtensionQuestions();
                    }
                }
            } catch (e) { console.warn('加载扩展题库失败', e); }
        }

        function saveExtensionQuestions() {
            const ext = state.allQuestions.filter(q => q.id > 200);
            try {
                localStorage.setItem('extensionQuestions', JSON.stringify(ext));
            } catch (e) { console.warn('保存扩展题库失败', e); }
        }

        // ---- 存储收藏/错题/掌握 ----
        function loadStorage() {
            try {
                const fav = localStorage.getItem('quizFav');
                if (fav) state.quizFav = new Set(JSON.parse(fav));
                const wrong = localStorage.getItem('quizWrong');
                if (wrong) state.quizWrong = new Set(JSON.parse(wrong));
                const mastered = localStorage.getItem('masteredIds');
                if (mastered) {
                    const ids = new Set(JSON.parse(mastered));
                    state.data.forEach(item => {
                        if (ids.has(item.id)) item.mastered = true;
                    });
                }
            } catch (e) { console.warn('Storage load error', e); }
        }

        function saveStorage() {
            try {
                localStorage.setItem('quizFav', JSON.stringify([...state.quizFav]));
                localStorage.setItem('quizWrong', JSON.stringify([...state.quizWrong]));
                const masteredIds = state.data.filter(d => d.mastered).map(d => d.id);
                localStorage.setItem('masteredIds', JSON.stringify(masteredIds));
                saveExtensionQuestions();
            } catch (e) { console.warn('Storage save error', e); }
        }

        // ---- 做题模块 ----
        function buildQuizList(filterType, isEn) {
            let pool = state.allQuestions.filter(q => q.lang === (isEn ? 'en' : 'zh'));
            if (filterType === 'fav') {
                pool = pool.filter(q => state.quizFav.has(q.id));
            } else if (filterType === 'wrong') {
                pool = pool.filter(q => state.quizWrong.has(q.id));
            }
            return pool;
        }

        function renderQuizCard(isEn) {
            const container = isEn ? 'enQuizCardArea' : 'quizCardArea';
            const progressId = isEn ? 'enQuizProgress' : 'quizProgress';
            const groupLabelId = isEn ? 'enQuizGroupLabel' : 'quizGroupLabel';
            const list = isEn ? state.enQuizQuestions : state.quizQuestions;
            const idx = isEn ? state.enQuizIndex : state.quizIndex;
            const groupIdx = isEn ? state.enQuizGroupIndex : state.quizGroupIndex;
            const groupSize = isEn ? state.enQuizGroupSize : state.quizGroupSize;
            const area = $(container);
            if (!area) return;
            if (list.length === 0) {
                area.innerHTML = `<div class="empty-state"><div class="icon">📭</div><h3>${isEn ? 'No questions' : '暂无题目'}</h3><p>${isEn ? 'Try changing filter' : '尝试切换筛选条件'}</p></div>`;
                $(progressId).textContent = `0 / 0`;
                $(groupLabelId).textContent = '0';
                return;
            }
            const totalGroups = Math.ceil(list.length / groupSize);
            const start = groupIdx * groupSize;
            const end = Math.min(start + groupSize, list.length);
            const groupQuestions = list.slice(start, end);
            const groupTotal = groupQuestions.length;
            if (groupTotal === 0 || idx >= groupTotal) {
                if (groupIdx < totalGroups - 1) {
                    if (isEn) state.enQuizGroupIndex++;
                    else state.quizGroupIndex++;
                    renderQuizCard(isEn);
                    return;
                } else {
                    area.innerHTML = `<div class="empty-state"><div class="icon">🎯</div><h3>${isEn ? 'All groups completed!' : '全部组完成！'}</h3></div>`;
                    $(progressId).textContent = `${end} / ${list.length}`;
                    $(groupLabelId).textContent = totalGroups;
                    return;
                }
            }
            const q = groupQuestions[idx];
            if (!q) {
                area.innerHTML = `<div class="empty-state"><div class="icon">🎯</div><h3>${isEn ? 'All done!' : '全部完成！'}</h3></div>`;
                $(progressId).textContent = `${end} / ${list.length}`;
                $(groupLabelId).textContent = `${groupIdx+1}/${totalGroups}`;
                return;
            }
            const total = list.length;
            $(progressId).textContent = `${end} / ${total}`;
            $(groupLabelId).textContent = `${groupIdx+1}/${totalGroups}`;
            const isAnswered = isEn ? state.enQuizAnswered : state.quizAnswered;
            const selectedOpt = isEn ? q._selected : q._selected;

                    <div class="q-text ${isEn ? 'en' : ''}">${q.q}</div>
                    <div class="options">
            `;
            const labels = isEn ? ['A','B','C','D'] : ['A','B','C','D'];
            q.options.forEach((opt, i) => {
                let cls = '', disabled = '', checked = '';
                if (isAnswered) {
                    if (i === q.answer) cls = 'correct';
                    else if (selectedOpt === i && i !== q.answer) cls = 'wrong';
                    disabled = 'disabled';
                } else if (selectedOpt === i) {
                    cls = 'selected';
                    checked = 'checked';
                }
                html += `<label class="${cls}" data-opt="${i}"><input type="radio" name="quiz_opt" value="${i}" ${checked} ${disabled} /><span>${labels[i]}. ${opt}</span></label>`;
            });
            html += `</div>`;
            if (isAnswered) {
                const isCorrect = selectedOpt === q.answer;
                html += `<div class="feedback show ${isCorrect ? 'correct' : 'wrong'}">${isCorrect ? '✅ 正确！' : '❌ 错误。正确答案是 ' + labels[q.answer]}<div class="exp">${q.explanation || ''}</div></div>`;
            }
            html += `
                </div>
                <div class="quiz-nav">
                    <div class="btn-group">
                        <button class="btn outline" id="${isEn ? 'enPrevQ' : 'prevQ'}">◀ 上一题</button>
                        <button class="btn outline" id="${isEn ? 'enNextQ' : 'nextQ'}">下一题 ▶</button>
                    </div>
                    <div class="btn-group">
                        <button class="btn ${isFav ? 'gold active' : 'outline'}" id="${isEn ? 'enFavQ' : 'favQ'}">⭐ ${isFav ? '已收藏' : '收藏'}</button>
                        <button class="btn primary-solid" id="${isEn ? 'enSubmitQ' : 'submitQ'}">${isAnswered ? '下一题' : '提交答案'}</button>
                    </div>
                </div>
            `;
            area.innerHTML = html;

                label.addEventListener('click', function(e) {
                    if (isAnswered) return;
                    const radio = this.querySelector('input[type="radio"]');
                    if (radio) radio.checked = true;
                    const idx2 = parseInt(this.dataset.opt);
                    if (isEn) {
                        state.enQuizQuestions[state.enQuizIndex + state.enQuizGroupIndex * state.enQuizGroupSize]._selected = idx2;
                    } else {
                        state.quizQuestions[state.quizIndex + state.quizGroupIndex * state.quizGroupSize]._selected = idx2;
                    }
                    renderQuizCard(isEn);
                });
            });

            const submitBtn = $(isEn ? 'enSubmitQ' : 'submitQ');
            if (submitBtn) {
                submitBtn.addEventListener('click', function() {
                    if (isAnswered) {
                        if (isEn) {
                            if (state.enQuizIndex < groupQuestions.length - 1) {
                                state.enQuizIndex++;
                                state.enQuizAnswered = false;
                            } else {
                                if (state.enQuizGroupIndex < totalGroups - 1) {
                                    state.enQuizGroupIndex++;
                                    state.enQuizIndex = 0;
                                    state.enQuizAnswered = false;
                                    renderQuizCard(true);
                                    return;
                                } else {
                                    showKnowledgeCard(isEn);
                                    return;
                                }
                            }
                        } else {
                            if (state.quizIndex < groupQuestions.length - 1) {
                                state.quizIndex++;
                                state.quizAnswered = false;
                            } else {
                                if (state.quizGroupIndex < totalGroups - 1) {
                                    state.quizGroupIndex++;
                                    state.quizIndex = 0;
                                    state.quizAnswered = false;
                                    renderQuizCard(false);
                                    return;
                                } else {
                                    showKnowledgeCard(isEn);
                                    return;
                                }
                            }
                        }
                        renderQuizCard(isEn);
                        return;
                    }
                    const qData = (isEn ? state.enQuizQuestions : state.quizQuestions)[
                        (isEn ? state.enQuizIndex : state.quizIndex) + (isEn ? state.enQuizGroupIndex : state.quizGroupIndex) * (isEn ? state.enQuizGroupSize : state.quizGroupSize)
                    ];
                    const selected = qData._selected;
                    if (selected === undefined || selected === null) {
                        alert(isEn ? 'Please select an option' : '请选择一个选项');
                        return;
                    }
                    const isCorrect = selected === qData.answer;
                    if (!isCorrect) {
                        state.quizWrong.add(qData.id);
                        const chapter = qData.chapter;
                        const relatedKnowledge = state.data.filter(k => k.chapter === chapter);
                        state.wrongKnowledge.push(...relatedKnowledge);
                        saveStorage();
                    }
                    if (isEn) state.enQuizAnswered = true;
                    else state.quizAnswered = true;
                    renderQuizCard(isEn);
                });
            }

            const prevBtn = $(isEn ? 'enPrevQ' : 'prevQ');
            const nextBtn = $(isEn ? 'enNextQ' : 'nextQ');
            if (prevBtn) {
                prevBtn.addEventListener('click', function() {
                    if (isEn) {
                        if (state.enQuizIndex > 0) { state.enQuizIndex--; state.enQuizAnswered = false; }
                        else if (state.enQuizGroupIndex > 0) { state.enQuizGroupIndex--; state.enQuizIndex = groupSize - 1; state.enQuizAnswered = false; }
                    } else {
                        if (state.quizIndex > 0) { state.quizIndex--; state.quizAnswered = false; }
                        else if (state.quizGroupIndex > 0) { state.quizGroupIndex--; state.quizIndex = groupSize - 1; state.quizAnswered = false; }
                    }
                    renderQuizCard(isEn);
                });
            }
            if (nextBtn) {
                nextBtn.addEventListener('click', function() {
                    if (isEn) {
                        if (state.enQuizIndex < groupQuestions.length - 1) { state.enQuizIndex++; state.enQuizAnswered = false; }
                        else if (state.enQuizGroupIndex < totalGroups - 1) { state.enQuizGroupIndex++; state.enQuizIndex = 0; state.enQuizAnswered = false; }
                    } else {
                        if (state.quizIndex < groupQuestions.length - 1) { state.quizIndex++; state.quizAnswered = false; }
                        else if (state.quizGroupIndex < totalGroups - 1) { state.quizGroupIndex++; state.quizIndex = 0; state.quizAnswered = false; }
                    }
                    renderQuizCard(isEn);
                });
            }

            const favBtn = $(isEn ? 'enFavQ' : 'favQ');
            if (favBtn) {
                favBtn.addEventListener('click', function() {
                    const id = qData.id;
                    if (state.quizFav.has(id)) state.quizFav.delete(id);
                    else state.quizFav.add(id);
                    saveStorage();
                    renderQuizCard(isEn);
                });
            }

                if (e.target.tagName === 'INPUT') return;
                if (e.key === ' ' || e.key === 'Space') { e.preventDefault(); if (submitBtn) submitBtn.click(); }
                if (e.key === 'ArrowLeft') { if (prevBtn) prevBtn.click(); }
                if (e.key === 'ArrowRight') { if (nextBtn) nextBtn.click(); }
            };
            document.removeEventListener('keydown', window._quizKeyHandler);
            window._quizKeyHandler = handleKey;
            document.addEventListener('keydown', handleKey);
        }

        // ---- 扩展坞功能（与原版相同） ----
        function initExtension() {
            const uploadZone = $('uploadZone');
            const fileInput = $('fileInput');
            const status = $('uploadStatus');
            uploadZone.addEventListener('click', () => fileInput.click());
            uploadZone.addEventListener('dragover', (e) => { e.preventDefault(); uploadZone.classList.add('dragover'); });
            uploadZone.addEventListener('dragleave', () => { uploadZone.classList.remove('dragover'); });
            uploadZone.addEventListener('drop', (e) => {
                e.preventDefault();
                uploadZone.classList.remove('dragover');
                if (e.dataTransfer.files.length) handleFile(e.dataTransfer.files[0]);
            });
            fileInput.addEventListener('change', (e) => {
                if (e.target.files.length) { handleFile(e.target.files[0]); fileInput.value = ''; }
            });

            function handleFile(file) {
                if (!file.name.endsWith('.txt')) {
                    status.innerHTML = `<span style="color:var(--red);">❌ 请上传 .txt 文件</span>`;
                    return;
                }
                const reader = new FileReader();
                reader.onload = function(e) {
                    const content = e.target.result;
                    const parsed = parseTxtQuestions(content);
                    if (parsed.length === 0) {
                        status.innerHTML = `<span style="color:var(--red);">❌ 解析失败，请检查格式（需包含 "### 题目：" 等标记）</span>`;
                        return;
                    }
                    let added = 0;
                    let maxId = Math.max(...state.allQuestions.map(q => q.id), 0);
                    parsed.forEach(q => {
                        const exists = state.allQuestions.some(ex => ex.q === q.q && ex.lang === q.lang);
                        if (!exists) {
                            q.id = ++maxId;
                            state.allQuestions.push(q);
                            added++;
                        }
                    });
                    if (added === 0) {
                        status.innerHTML = `<span style="color:var(--gold);">⚠️ 所有题目已存在，未新增</span>`;
                    } else {
                        saveExtensionQuestions();
                        status.innerHTML = `<span style="color:var(--green);">✅ 成功导入 ${added} 道题目！</span>`;
                        updateStats();
                        if (document.querySelector('.tab-btn[data-tab="quiz"]').classList.contains('active')) {
                            state.quizQuestions = buildQuizList(state.quizFilter, false);
                            state.quizIndex = 0; state.quizAnswered = false; state.quizGroupIndex = 0;
                            renderQuizCard(false);
                        }
                        if (document.querySelector('.tab-btn[data-tab="enquiz"]').classList.contains('active')) {
                            state.enQuizQuestions = buildQuizList(state.enQuizFilter, true);
                            state.enQuizIndex = 0; state.enQuizAnswered = false; state.enQuizGroupIndex = 0;
                            renderQuizCard(true);
                        }
                        renderReviewList();
                    }
                };
                reader.onerror = function() { status.innerHTML = `<span style="color:var(--red);">❌ 读取文件失败</span>`; };
                reader.readAsText(file, 'UTF-8');
            }

            function parseTxtQuestions(text) {
                const lines = text.split(/\r?\n/);
                const questions = [];
                let current = null, options = [], answer = -1, explanation = '', chapter = '未知章节', lang = 'zh';
                for (let i = 0; i < lines.length; i++) {
                    let line = lines[i].trim();
                    if (!line) continue;
                    if (line.startsWith('### 题目：') || line.startsWith('### 题目：')) {
                        if (current && options.length === 4 && answer >= 0) {
                            questions.push({ id:0, chapter, lang, q: current, options, answer, explanation: explanation || '无解析' });
                        }
                        current = line.replace(/^###\s*题目[：:]\s*/, '').trim();
                        options = []; answer = -1; explanation = '';
                        lang = /[\u4e00-\u9fa5]/.test(current) ? 'zh' : 'en';
                        const chapterKeywords = ['头部','颈部','胸部','腹部','盆部','脊柱','上肢','下肢','Head','Neck','Thorax','Abdomen','Pelvis','Spine','Upper','Lower'];
                        for (let kw of chapterKeywords) { if (current.includes(kw)) { chapter = kw; break; } }
                        continue;
                    }
                    const optMatch = line.match(/^([A-D])\.\s*(.*)/);
                    if (optMatch) { options.push(optMatch[1] + '. ' + optMatch[2]); continue; }
                    const ansMatch = line.match(/^答案[：:]\s*([A-D])/);
                    if (ansMatch) { answer = ansMatch[1].charCodeAt(0) - 65; continue; }
                    const expMatch = line.match(/^解析[：:]\s*(.*)/);
                    if (expMatch) { explanation = expMatch[1]; continue; }
                    if (line.startsWith('---')) continue;
                    if (current !== null) current += ' ' + line;
                }
                if (current && options.length === 4 && answer >= 0) {
                    questions.push({ id:0, chapter, lang, q: current, options, answer, explanation: explanation || '无解析' });
                }
                return questions;
            }
        }
